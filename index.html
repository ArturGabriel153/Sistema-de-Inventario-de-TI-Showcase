Sobre o Projeto

Gestão de TI para múltiplas filiais

Inventário TI v2.0 é uma plataforma full-stack para equipes de TI gerenciarem todo o parque tecnológico de uma rede de filiais — equipamentos, manutenções, coletores, tablets, periféricos e depósito em um único sistema.

🔍

Problema resolvido

Controlar centenas de ativos de TI espalhados por dezenas de filiais, sem histórico de manutenção e sem visibilidade de quais equipamentos precisam de atenção imediata.

⚡

Prioridade automática

O sistema calcula automaticamente a prioridade de manutenção de cada equipamento com base na data da última intervenção — OK (<4 meses), Atenção (4–5 meses), Crítico (≥6 meses ou nunca mantido).

📄

Relatórios PDF & XLSX

Exportação via window.print() (sem biblioteca extra) para PDF e via biblioteca xlsx para planilhas — com histórico completo de manutenções e fotos por equipamento.

// backend/src/controllers/equipamentosController.js
// Cálculo automático de prioridade baseado na última manutenção
// Executado para cada equipamento no dashboard e na listagem

function calcularPrioridade(equip, ultimaManutencaoData) {
  // Impressoras não entram no sistema de prioridade
  if (equip.tipo === 'impressora') {
    return { ...equip, prioridade: null, meses_sem_manutencao: null };
  }

  // Nunca recebeu manutenção → imediatamente Crítico
  if (!ultimaManutencaoData) {
    return { ...equip, prioridade: 'Crítico', meses_sem_manutencao: null };
  }

  const meses = Math.floor(
    (Date.now() - new Date(ultimaManutencaoData).getTime())
    / (1000 * 60 * 60 * 24 * 30)
  );

  // Regras de negócio:
  // OK       → menos de 4 meses
  // Atenção  → 4 ou 5 meses
  // Crítico  → 6 meses ou mais
  let prioridade = 'OK';
  if      (meses >= 6) prioridade = 'Crítico';
  else if (meses >= 4) prioridade = 'Atenção';

  return { ...equip, prioridade, meses_sem_manutencao: meses, ultima_manutencao: ultimaManutencaoData };
}

// backend/src/middlewares/authMiddleware.js
// Verifica o Firebase ID Token no header Authorization
// e injeta req.user = { uid, email, name } em todos os controllers

const admin = require('../config/firebase');

async function authMiddleware(req, res, next) {
  const header = req.headers.authorization;
  if (!header?.startsWith('Bearer ')) {
    return res.status(401).json({ error: 'Token não fornecido' });
  }
  try {
    const token   = header.split('Bearer ')[1];
    const decoded = await admin.auth().verifyIdToken(token);
    req.user = { uid: decoded.uid, email: decoded.email, name: decoded.name };
    next();
  } catch {
    return res.status(401).json({ error: 'Token inválido' });
  }
}
